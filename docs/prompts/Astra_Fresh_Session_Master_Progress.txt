def execute_window(
    oos_candidates: list[dict],
    capital: float,
    base_risk: float,
) -> dict:
    if not oos_candidates:
        return dict(trades=0, wins=0, win_rate=0.0, net_pnl=0.0, max_dd=0.0, equity=[capital])

    oos_sorted = sorted(oos_candidates, key=lambda c: c["ts_ms"])

    equity     = capital
    peak       = capital
    max_dd     = 0.0
    trades     = 0
    wins       = 0
    equity_curve = [capital]

    TRADE_DURATION_MS = TRADE_BARS * 15 * 60 * 1000
    active_exit_ms: list[int] = []

    for c in oos_sorted:
        entry_ms = c["ts_ms"]
        exit_ms  = entry_ms + TRADE_DURATION_MS

        active_exit_ms = [e for e in active_exit_ms if e > entry_ms]

        if len(active_exit_ms) >= MAX_CONCURRENT:
            continue

        elif equity - capital >= 50.0:
            risk_usd = base_risk * 1.50
        else:
            risk_usd = base_risk

        net_pnl = c["outcome_r"] * risk_usd

        equity += net_pnl
        trades += 1
        if net_pnl > 0:
            wins += 1

        if equity > peak:
            peak = equity
        dd = (peak - equity) / peak * 100.0
        if dd > max_dd:
            max_dd = dd

        equity_curve.append(equity)
        active_exit_ms.append(exit_ms)

            break

        trades     = trades,
        wins       = wins,
        win_rate   = win_rate,
        net_pnl    = net_pnl,
        max_dd     = max_dd,
